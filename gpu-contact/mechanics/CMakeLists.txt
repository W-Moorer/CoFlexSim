cmake_minimum_required(VERSION 3.22)
project(coflex_mechanics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
if(NOT EXISTS "${EIGEN_ROOT}/Eigen/Core")
  message(FATAL_ERROR "Set EIGEN_ROOT to Eigen 3.4.0 headers")
endif()
foreach(name IN ITEMS coflex_native coflex_pd_native coflex_mortar_native)
  if(name STREQUAL "coflex_mortar_native")
    pybind11_add_module(${name} mortar_consistent_native.cpp)
  else()
    pybind11_add_module(${name} native.cpp)
    target_compile_definitions(${name} PRIVATE COFLEX_MODULE_NAME=${name})
  endif()
  target_include_directories(${name} PRIVATE "${EIGEN_ROOT}")
  target_compile_definitions(${name} PRIVATE EIGEN_DONT_PARALLELIZE)
  target_link_libraries(${name} PRIVATE OpenMP::OpenMP_CXX)
  if(MSVC)
    target_compile_options(${name} PRIVATE /O2 /bigobj)
  else()
    target_compile_options(${name} PRIVATE -O3)
  endif()
endforeach()
