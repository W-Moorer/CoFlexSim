cmake_minimum_required(VERSION 3.22)
project(coflex_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CUDAToolkit 12.6 REQUIRED)
add_executable(coflex_compile src/compile_cuda.cpp)
target_link_libraries(coflex_compile PRIVATE CUDA::nvrtc)
set(NATIVE_IMAGES)
foreach(arch IN ITEMS sm_86 sm_89 compute_86)
  set(ext cubin)
  if(arch MATCHES "compute")
    set(ext ptx)
  endif()
  set(out "${CMAKE_CURRENT_BINARY_DIR}/images/region_${arch}.${ext}")
  add_custom_command(OUTPUT "${out}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/images"
    COMMAND coflex_compile "${CMAKE_CURRENT_SOURCE_DIR}/kernels/region.cu" "${out}" ${arch}
    DEPENDS coflex_compile kernels/region.cu VERBATIM)
  list(APPEND NATIVE_IMAGES "${out}")
endforeach()
add_custom_target(coflex_images ALL DEPENDS ${NATIVE_IMAGES})
add_library(coflex_native SHARED src/region.cpp src/scheduler.cpp)
target_include_directories(coflex_native PUBLIC include)
target_link_libraries(coflex_native PRIVATE CUDA::cuda_driver CUDA::cublas)
target_compile_definitions(coflex_native PRIVATE COFLEX_EXPORTS)
if(MSVC)
  target_compile_options(coflex_native PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(coflex_native PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_dependencies(coflex_native coflex_images)
